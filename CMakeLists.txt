cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
project(URSoccerLabClient VERSION 0.1.1 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(URS_BUILD_CONNECTOR "Build the C++ TCP/video connector" ON)
option(URS_BUILD_VIEWER "Build the SDL desktop viewer" ON)
if(URS_BUILD_VIEWER AND NOT URS_BUILD_CONNECTOR)
  message(FATAL_ERROR "The viewer requires URS_BUILD_CONNECTOR=ON")
endif()
include(GNUInstallDirs)
include(CTest)
add_library(urs_protocol STATIC src/protocol.c)
add_library(URSoccerLab::protocol ALIAS urs_protocol)
set_target_properties(urs_protocol PROPERTIES EXPORT_NAME protocol)
target_include_directories(urs_protocol PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT WIN32)
  target_link_libraries(urs_protocol PUBLIC m)
endif()
if(BUILD_TESTING)
  add_executable(protocol_test tests/protocol_test.c)
  target_link_libraries(protocol_test PRIVATE urs_protocol)
  add_test(NAME protocol COMMAND protocol_test)
endif()
if(URS_BUILD_CONNECTOR)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
  find_package(Threads REQUIRED)
  include(FetchContent)
  FetchContent_Declare(cjson URL https://github.com/DaveGamble/cJSON/archive/refs/tags/v1.7.19.tar.gz
    URL_HASH SHA256=7fa616e3046edfa7a28a32d5f9eacfd23f92900fe1f8ccd988c1662f30454562
    SOURCE_SUBDIR no-project)
  FetchContent_MakeAvailable(cjson)
  add_library(urs_json STATIC ${cjson_SOURCE_DIR}/cJSON.c)
  target_include_directories(urs_json PUBLIC $<BUILD_INTERFACE:${cjson_SOURCE_DIR}>)
  add_library(urs_client STATIC src/client.cpp src/reply.cpp)
  add_library(URSoccerLab::client ALIAS urs_client)
  set_target_properties(urs_client PROPERTIES EXPORT_NAME client)
  target_include_directories(urs_client PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_libraries(urs_client PUBLIC urs_protocol PkgConfig::FFMPEG Threads::Threads PRIVATE urs_json)
  if(WIN32)
    target_link_libraries(urs_client PUBLIC ws2_32)
  endif()
  if(BUILD_TESTING)
    add_executable(reply_test tests/reply_test.cpp)
    target_link_libraries(reply_test PRIVATE urs_client)
    add_test(NAME readable_replies COMMAND reply_test)
    add_executable(client_probe tests/client_probe.cpp)
    target_link_libraries(client_probe PRIVATE urs_client)
  endif()
  install(TARGETS urs_client urs_json EXPORT URSClientTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(DIRECTORY include/ursoccerlab DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  install(FILES ${cjson_SOURCE_DIR}/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/ursoccerlab-client/licenses RENAME cJSON-LICENSE)
endif()
if(URS_BUILD_VIEWER)
  find_package(SDL2 CONFIG REQUIRED)
  FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/refs/tags/v1.91.9b.tar.gz
    URL_HASH SHA256=8e1bbc76c71d74fef2fb85db7e7ca8eba13d6a86623c54992b60162db554ffdb)
  FetchContent_MakeAvailable(imgui)
  add_library(urs_imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer2.cpp)
  target_include_directories(urs_imgui PUBLIC ${imgui_SOURCE_DIR})
  target_link_libraries(urs_imgui PUBLIC SDL2::SDL2)
  add_executable(urs-viewer src/main.cpp)
  target_link_libraries(urs-viewer PRIVATE urs_client urs_imgui)
  if(BUILD_TESTING)
    add_test(NAME gui_initialization COMMAND urs-viewer --smoke-test)
    set_tests_properties(gui_initialization PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 15)
  endif()
  install(TARGETS urs-viewer RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES ${imgui_SOURCE_DIR}/LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/ursoccerlab-client/licenses RENAME ImGui-LICENSE)
endif()
install(TARGETS urs_protocol EXPORT URSClientTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/urs_protocol.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT URSClientTargets NAMESPACE URSoccerLab:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/URSoccerLabClient)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/URSoccerLabClientConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/URSoccerLabClientConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/URSoccerLabClient)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/URSoccerLabClientConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/URSoccerLabClientConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/URSoccerLabClientConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/URSoccerLabClient)
install(FILES LICENSE NOTICE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/ursoccerlab-client)
